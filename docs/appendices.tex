\chapter{Dataset sources}
\label{apx:dataset_sources}
Following are links to all datasets that were used in construction of dataset used for the purposes of this thesis.

\begin{itemize}
    \item \textbf{Speech:}
    \begin{itemize}
        \item MLCommons Peoples speech: \url{https://hf.co/datasets/MLCommons/peoples_speech}
        \item English--Arabic Speech Translation (English subset): \url{https://hf.co/datasets/ammagra/english-arabic-speech-translation}
        \item ATCO2 Corpus: \url{https://hf.co/datasets/Jzuluaga/atco2_corpus_1h}
        \item NOIZEUS: \url{https://ecs.utdallas.edu/loizou/speech/noizeus/}
    \end{itemize}

    \item \textbf{Music:}
    \begin{itemize}
        \item MusicGen Fine-tuning (Jazz, Electronic, Classical): \url{https://hf.co/datasets/PlutoG99001/}
        \item Lewtun Music Genres: \url{https://hf.co/datasets/lewtun/music_genres}
        \item Ylacombe Music Genres Country: \url{https://hf.co/datasets/ylacombe/music_genres_Country}
        \item POPMUSIC1981: \url{https://hf.co/datasets/memepottaboah/POPMUSIC1981}
        \item CCMusic Acapella: \url{https://hf.co/datasets/ccmusic-database/acapella}
    \end{itemize}

    \item \textbf{Mixed:}
    \begin{itemize}
        \item MUSAN: \url{https://www.openslr.org/17/}
    \end{itemize}
\end{itemize}
